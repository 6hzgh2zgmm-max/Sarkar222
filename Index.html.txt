<script>
const members = {
  "S001": {name: "Ali Bhatti", photo: "S001.jpg"},
  "S002": {name: "Zain", photo: "S002.jpg"},
  "S003": {name: "Azeem", photo: "S003.jpg"},
  "S004": {name: "Zahoor", photo: "S004.jpg"},
  "S005": {name: "Faisal", photo: "S005.jpg"}
};

function searchMember(){
  let code = document.getElementById("code").value.toUpperCase();
  let member = members[code];
  let resultDiv = document.getElementById("result");

  if(member){
    let link = window.location.origin + window.location.pathname + "?id=" + code;
    resultDiv.innerHTML = `
      <div class="card">
        <h2>${member.name}</h2>
        <p><b>Code:</b> ${code}</p>
        <img src="${member.photo}">
        <p><b>QR Code:</b></p>
        <canvas id="qrcode"></canvas>
      </div>
    `;
    QRCode.toCanvas(document.getElementById('qrcode'), link, {width: 150});
  } else {
    resultDiv.innerHTML = "<p style='color:red'>Code nahi mila!</p>";
  }
}
</script>
